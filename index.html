Kód pro Cloudflare Pages Function

Uložte tento kód do souboru functions/api/fve-status.js ve vašem projektu. Serverless funkce zavolá Shelly Cloud s vaším tajným klíčem a data pošle do tohoto dashboardu: 

/**
 * Cloudflare Pages Function: /api/fve-status
 * Uložte jako: functions/api/fve-status.js
 */
export async function onRequestGet(context) {
  const { env } = context;

  const authKey = env.SHELLY_AUTH_KEY;
  // Váš dedikovaný server Shelly Cloud:
  const serverUrl = env.SHELLY_SERVER_URL || 'https://shelly-137-eu.shelly.cloud';
  const deviceIdsRaw = env.SHELLY_DEVICE_IDS || '';

  if (!authKey) {
    return new Response(JSON.stringify({ 
      error: 'SHELLY_AUTH_KEY není nastaven v Cloudflare Environment Secrets.' 
    }), { 
      status: 500,
      headers: { 'Content-Type': 'application/json' }
    });
  }

  const deviceIds = deviceIdsRaw.split(',').map(s => s.trim()).filter(Boolean);
  const baseUrl = serverUrl.replace(/\/$/, '');
  const results = {};

  try {
    for (const devId of deviceIds) {
      const url = `${baseUrl}/device/status?id=${encodeURIComponent(devId)}&auth_key=${encodeURIComponent(authKey)}`;
      const res = await fetch(url, { headers: { 'User-Agent': 'Cloudflare-Pages-FVE/1.0' } });
      if (res.ok) {
        const json = await res.json();
        if (json.isok && json.data) {
          results[devId] = json.data.device_status;
        }
      }
    }

    return new Response(JSON.stringify({ 
      success: true, 
      timestamp: Date.now(),
      server: baseUrl,
      devices: results 
    }), {
      headers: { 
        'Content-Type': 'application/json',
        'Cache-Control': 'no-store'
      }
    });
  } catch (err) {
    return new Response(JSON.stringify({ error: err.message }), { 
      status: 502,
      headers: { 'Content-Type': 'application/json' }
    });
  }
}

V Cloudflare administraci nastavíte: SHELLY_SERVER_URL = https://shelly-137-eu.shelly.cloud, SHELLY_AUTH_KEY (zaškrtnout Encrypt/Secret) a SHELLY_DEVICE_IDS.

Nastavení připojení k Shelly Cloud

Bezpečný serverless režim: Dashboard se dotazuje na interní koncový bod /api/fve-status. Váš tajný Shelly Auth Key je bezpečně uložen v Cloudflare Secrets a na frontend se nikdy neodešle.
⚠️ V přímém režimu volá prohlížeč Shelly Cloud přímo. Auth klíč se ukládá pouze v localStorage vašeho prohlížeče.
